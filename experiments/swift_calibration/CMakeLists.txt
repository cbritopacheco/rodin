# Calibration drivers are explicit targets, not examples built by default.
foreach(family IN ITEMS Reconstruction Sweep)
  foreach(dimension IN ITEMS 2 3)
    if(family STREQUAL Reconstruction)
      set(source Reconstruction${dimension}D.cpp)
    else()
      set(source Trajectory${dimension}D.cpp)
    endif()
    if(dimension EQUAL 2)
      set(suffix "")
    else()
      set(suffix 3D)
    endif()
    foreach(degree IN ITEMS 1 2)
      set(target LevelSetSWIFT${family}${suffix})
      if(degree EQUAL 2)
        string(APPEND target P2)
      endif()
      add_executable(${target} EXCLUDE_FROM_ALL ${source})
      if(degree EQUAL 2)
        target_compile_definitions(${target} PRIVATE RODIN_SWIFT_P2_DISPLACEMENT=1)
      endif()
      target_link_libraries(${target}
        PRIVATE Rodin::Geometry Rodin::Solver Rodin::Variational Rodin::MMG)
    endforeach()
  endforeach()
endforeach()
