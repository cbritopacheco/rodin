cmake_minimum_required(VERSION 3.16)
project(RodinInstallationTest LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find the installed Rodin package
find_package(Rodin REQUIRED)

# Create test executable
add_executable(test_app main.cpp)

# Ensure the configured package and the installed header describe one version.
target_compile_definitions(test_app PRIVATE
  EXPECTED_RODIN_VERSION="${Rodin_VERSION}"
)

# Link against Rodin libraries
target_link_libraries(test_app PRIVATE
  Rodin::Geometry
  Rodin::Variational
  Rodin::Solver
)
