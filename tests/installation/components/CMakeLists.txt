cmake_minimum_required(VERSION 3.16)
project(RodinComponentConsumer LANGUAGES C CXX)

find_package(Rodin REQUIRED COMPONENTS ${RODIN_COMPONENT})
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_20)
target_link_libraries(consumer PRIVATE Rodin::${RODIN_COMPONENT})
if(RODIN_COMPONENT STREQUAL "Kokkos::MPI")
  target_compile_definitions(consumer PRIVATE RODIN_CONSUMER_MPI)
elseif(RODIN_COMPONENT STREQUAL "Kokkos::PETSc")
  target_compile_definitions(consumer PRIVATE RODIN_CONSUMER_PETSC)
elseif(RODIN_COMPONENT STREQUAL "Kokkos")
  target_compile_definitions(consumer PRIVATE RODIN_CONSUMER_KOKKOS)
endif()
if(CMAKE_DISABLE_FIND_PACKAGE_PETSc AND TARGET PETSc::PETSc)
  message(FATAL_ERROR "The selected component discovered PETSc unexpectedly")
endif()
